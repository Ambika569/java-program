OSGI Components and Services :- 

Simple Java CLasses. 

OSGI R7 annotations:- 

@Component:- To register a normal java class as a OSGI Component
import org.osgi.service.component.annotations.Component;

@Activate :- import org.osgi.service.component.annotations.Activate;
@Modified :- import org.osgi.service.component.annotations.Modified;
@DeActivate :- import org.osgi.service.component.annotations.DeActivate;

once you have deployed the code ---> Activate
Again if you deploy the code  --> Deactivated and then Activated



=====================================


@Component(immediate = true)
public class Test{

private static final Logger log = LoggerFactory.getLogger(Test.class); //import org.slf4j.Logger

//import org.slf4j.LoggerFactory;

@Activate
public void active(){
log.info("Test Class ---- Activated...");

}

@DeActivate
public void deActive(){
log.info("Test Class ---- DeActivated...");
}

}

package com.karthik.newsportal.core.services;

import org.osgi.service.component.annotations.Component;
import org.osgi.service.component.annotations.Deactivate;
import org.osgi.service.component.annotations.Modified;
import org.slf4j.LoggerFactory;
import org.osgi.service.component.annotations.Activate;
import org.slf4j.Logger;


@Component(immediate = true)
public class Test {

	private static final Logger log = LoggerFactory.getLogger(Test.class);
	
	@Activate
	public void active() {
		log.info("XXXXXXXX is Activated.....");
		
	}
	
	@Deactivate
public void deactive() {
		log.info("XXXXXXX is DeActivated.....");
	}
@Modified
public void modified() {
	log.info("XXXXXXX is Modified.....");
}
}
